# My Own Result (2026-08-06)

```ts
type Result<TData = void, TError = string> = { ok: false; error: TError } | { ok: true; data: TData }

const voidValue = undefined as unknown as void

const ok = <TData, TError>(data: TData): Result<TData, TError> => ({
  ok: true,
  data,
})

ok.void = () => ok(voidValue)

const err = <TError>(error: TError): Result<never, TError> => ({
  ok: false,
  error,
})

const unwrapOr = <TData, TValue>(result: Result<TData>, callback: (data: TData) => TValue, fallback: TValue) =>
  result.ok ? callback(result.data) : fallback

const convertResult = <TInput, TOutput, TError>(
  result: Result<TInput, TError>,
  convert: (data: TInput) => TOutput,
): Result<TOutput, TError> => (result.ok ? ok(convert(result.data)) : err(result.error))
```
